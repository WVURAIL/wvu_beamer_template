\section{Motivation}
\begin{frame}{The question}
  State the question your audience should understand.
  \begin{itemize}
    \item Why does it matter?
    \item What is already known?
    \item What will this presentation add?
  \end{itemize}
\end{frame}

\section{Results}
\begin{frame}{The key result}
  \begin{block}{Main finding}
    Put the central result here. Add a figure, equation, or comparison
    that supports it.
  \end{block}
  \begin{information}[Context]
    Explain the conditions needed to interpret the result.
  \end{information}
\end{frame}

\section{Takeaways}
\begin{frame}{What to remember}
  \begin{enumerate}
    \item Summarize the finding.
    \item Explain why it matters.
    \item Identify the next step.
  \end{enumerate}
\end{frame}
