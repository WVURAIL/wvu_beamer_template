% Compact previews of the navy panel used by title and divider slides.
% Keep the texture scale, crop, opacity, fade, and rule matched to the theme.
\makeatletter
\newcommand{\wvupatternsample}[1]{%
  \begingroup
  \wvupattern{#1}%
  \resizebox{\linewidth}{!}{%
    \begin{tikzpicture}
      \path[use as bounding box] (0,0)
        rectangle (\paperwidth,.5\paperheight+15pt);
      \clip (0,0) rectangle (\paperwidth,.5\paperheight+15pt);
      \fill[wvu-blue] (0,0)
        rectangle (\paperwidth,.5\paperheight+15pt);
      \begin{scope}
        \clip (0,0) rectangle (.42\paperwidth,.5\paperheight+15pt);
        \ifdefstring{\wvu@pattern}{none}{}{%
          \edef\wvu@pattern@file{\csuse{wvu@pattern@file@\wvu@pattern}}%
          \def\wvu@pattern@size{width=.60\paperwidth}%
          \node[anchor=south west,inner sep=0pt,opacity=.14]
            at (-.08\paperwidth,-.12\paperheight)
            {\wvu@pattern@image};
        }%
        \fill[wvu-blue,path fading=west] (.24\paperwidth,0)
          rectangle (.42\paperwidth,.5\paperheight+15pt);
      \end{scope}
      \fill[wvu-gold] (0,.5\paperheight+10pt)
        rectangle (\paperwidth,.5\paperheight+15pt);
    \end{tikzpicture}%
  }%
  \endgroup
}
\makeatother

\begin{frame}{\wvuheading{Keep or turn off the pattern}}
  Topo appears automatically on title, section, and closing slides.
  \bigskip

  \begin{columns}[T,onlytextwidth]
    \begin{column}{.47\textwidth}
      {\large\bfseries\color{wvu-blue}Topo - Solid\par}
      \smallskip
      \wvupatternsample{topo-solid}
      \smallskip
      {\footnotesize\texttt{\string\wvusetup\{pattern=topo-solid\}}\par}
      \medskip
      The default. No extra setup is needed.
    \end{column}
    \begin{column}{.47\textwidth}
      {\large\bfseries\color{wvu-blue}No pattern\par}
      \smallskip
      \wvupatternsample{none}
      \smallskip
      {\footnotesize\texttt{\string\wvusetup\{pattern=none\}}\par}
      \medskip
      Use a plain panel throughout the presentation.
    \end{column}
  \end{columns}

  \bigskip
  {\small Put the setting in your preamble. Both options leave content slides
  unchanged.}
\end{frame}
